cmake_minimum_required(VERSION 3.10)
project(vnc_mp1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(VNC_BUILD_SDN "Build CODAC Linux SDN application" ON)
include(CTest)
add_library(vnc-processing Processing.cpp)
target_include_directories(vnc-processing PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(BUILD_TESTING)
    add_executable(processing-test tests/ProcessingTest.cpp)
    target_link_libraries(processing-test PRIVATE vnc-processing)
    add_test(NAME processing COMMAND processing-test)
endif()

if(VNC_BUILD_SDN)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "SDN application requires CODAC Linux; use -DVNC_BUILD_SDN=OFF for algorithm tests")
    endif()
    set(CODAC_ROOT "$ENV{CODAC_ROOT}" CACHE PATH "CODAC installation")
    if(NOT CODAC_ROOT)
        set(CODAC_ROOT /opt/codac)
    endif()
    find_package(LibXml2 REQUIRED)
    find_package(Threads REQUIRED)
    find_path(SDN_INCLUDE_DIR sdn-core/sdn-api.h HINTS "${CODAC_ROOT}/include")
    foreach(component sdn-core ccs-core log rt)
        string(REPLACE "-" "_" variable "${component}_LIBRARY")
        find_library(${variable} NAMES ${component} HINTS "${CODAC_ROOT}/lib")
        if(NOT ${variable})
            message(FATAL_ERROR "Missing CODAC dependency: ${component}")
        endif()
        list(APPEND CODAC_LIBRARIES "${${variable}}")
    endforeach()
    if(NOT SDN_INCLUDE_DIR)
        message(FATAL_ERROR "Missing CODAC SDN headers")
    endif()
    add_executable(vnc-mp1 main.cpp SdnTransport.cpp)
    target_include_directories(vnc-mp1 PRIVATE "${SDN_INCLUDE_DIR}" ${LIBXML2_INCLUDE_DIRS} ${LIBXML2_INCLUDE_DIR})
    target_link_libraries(vnc-mp1 PRIVATE vnc-processing ${CODAC_LIBRARIES} ${LIBXML2_LIBRARIES} Threads::Threads)
    set_target_properties(vnc-mp1 PROPERTIES INSTALL_RPATH "${CODAC_ROOT}/lib")
    install(TARGETS vnc-mp1 RUNTIME DESTINATION bin)
    set(VNC_TOPIC_INSTALL_DIR etc/sdn CACHE STRING "Topic directory relative to install prefix, or absolute")
    set(topic_files "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/vnc-mp1-topic.xml")
    foreach(i RANGE 0 5)
        list(APPEND topic_files "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/pcf${i}-nds-sdn-topic.xml")
    endforeach()
    install(FILES ${topic_files} DESTINATION "${VNC_TOPIC_INSTALL_DIR}")
endif()
